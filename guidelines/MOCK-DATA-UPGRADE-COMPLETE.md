### **Example 2: Blog Posts by Author**

```typescript

console.log(sarahPosts.length); // 24 posts

// Get author details
const sarah = blogAuthors.find(a => a.slug === 'sarah-mitchell');

```

### **Example 3: Navigation Menu**

```typescript
import { mainNavigation } from '../data';

// Render navigation
mainNavigation.map(item => (
  <MenuItem key={item.page}>
    {item.label}
    {item.children && (
      <Submenu>
        {item.children.map(child => (
          <SubMenuItem key={child.page}>
            {child.label}
          </SubMenuItem>
        ))}
      </Submenu>
    )}
  </MenuItem>
));
```
